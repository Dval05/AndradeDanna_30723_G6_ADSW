package "Capa de Servicios (Lógica de Negocio)" {

        - usuarioDAO: IUsuarioDAO

' ============ CAPA DAO ============
package "Capa DAO (Persistencia)" {
    interface IEmpleadoDAO {
        + crear(Empleado)
        + buscarPorCedula(cedula): Empleado
        + listarTodos(): List
        + editar(Empleado)
        + eliminar(cedula)
    }
    
    interface IUsuarioDAO {
        + crear(Usuario)
        + buscarPorUsuario(usuario): Usuario
        + buscarPorCorreo(correo): Usuario
        + editar(Usuario)
    }
    
    interface IReporteDAO {
        + crear(Reporte)
        + buscarPorFecha(inicio, fin): List
        + listarTodos(): List
    }
    
    class EmpleadoDAO {
        - conexion: ConexionBD
        + crear(Empleado)
        + buscarPorCedula(cedula): Empleado
        + listarTodos(): List
        + editar(Empleado)
        + eliminar(cedula)
    }
    
    class UsuarioDAO {
        - conexion: ConexionBD
        + crear(Usuario)
        + buscarPorUsuario(usuario): Usuario
        + buscarPorCorreo(correo): Usuario
        + editar(Usuario)
    }
    
    class ReporteDAO {
        - conexion: ConexionBD
        + crear(Reporte)
        + buscarPorFecha(inicio, fin): List
        + listarTodos(): List
    }
    
    IEmpleadoDAO <|.. EmpleadoDAO
    IUsuarioDAO <|.. UsuarioDAO
    IReporteDAO <|.. ReporteDAO

' ============ PATRONES DE DISEÑO ============
package "Patrones de Diseño" {
    interface IComando {
        + ejecutar(): ResultadoComando
        + deshacer()
    }
    
    class CrearEmpleadoComando {
        - empleado: Empleado
        - servicio: EmpleadoServicio
        + ejecutar(): ResultadoComando
        + deshacer()
    }
    
    class InvocadorOperaciones {
        - historialOperaciones: Stack
        + ejecutar(IComando): ResultadoComando
        + deshacerUltima()
        + rehacer()
    }
    
    interface ISujeto {
        + agregarObservador(IObservador)
        + eliminarObservador(IObservador)
        + notificarObservadores(EventoSistema)
    }
    
    interface IObservador {
        + actualizar(EventoSistema)
        + getNombre(): String
    }
    
    class SujetoGestionEmpleado {
        - observadores: List
        + agregarObservador(IObservador)
        + eliminarObservador(IObservador)
        + notificarObservadores(EventoSistema)
    }
    
    class ObservadorAuditoria {
        + actualizar(EventoSistema)
        + getNombre(): String
        - registrarEnBD(String)
    }
    
    class ObservadorNotificacion {
        + actualizar(EventoSistema)
        + getNombre(): String
        - enviarNotificacion(String, String)
    }
    
    IComando <|.. CrearEmpleadoComando
    ISujeto <|.. SujetoGestionEmpleado
    IObservador <|.. ObservadorAuditoria
    IObservador <|.. ObservadorNotificacion
}

@startuml Secuencia_Autenticacion

title Secuencia: Iniciar Sesión\nRF-02.01 - SNAAR TekMess

actor Usuario
participant "login.jsp" as login
participant "AuthController" as controller
participant "AuthServicio" as servicio
participant "UsuarioDAO" as dao
participant "CifradorContrasena" as cifrador
participant "BaseDatos" as bd
participant "SujetoGestionEmpleado" as sujeto
participant "ObservadorAuditoria" as auditoria

Usuario -> login: Ingresa usuario y contraseña
login -> controller: POST /api/auth/login
controller -> servicio: autenticar(usuario, contraseña)

servicio -> dao: buscarPorUsuario(usuario)
dao -> bd: SELECT FROM usuarios WHERE usuario=?
bd --> dao: Usuario encontrado

servicio -> cifrador: verificar(contraseña, hash)
cifrador --> servicio: true/false

alt Credenciales válidas
    servicio -> servicio: crearSesion(usuario)
    servicio -> sujeto: notificarObservadores(evento)
    sujeto -> auditoria: actualizar(evento)
    auditoria -> bd: INSERT auditoria
    servicio --> controller: Sesion activa
    controller --> login: HTTP 200 + Cookie sesión
    login --> Usuario: Redirect al dashboard
else Credenciales inválidas
    servicio -> sujeto: notificarObservadores(intento_fallido)
    sujeto -> auditoria: registrarIntentofallido()
    servicio --> controller: HTTP 401 Unauthorized
    controller --> login: Mostrar error
    login --> Usuario: "Credenciales inválidas"
end

@enduml
